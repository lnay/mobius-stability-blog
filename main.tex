\section{Main statements}

\begin{theorem}
	There is a group action:
	\[
		\aut\left(\triangcat\right)^{\mathrm{num}}
		\circlearrowright
		%\HCode{<mo class="MathClass-rel" stretchy="false">⟳</mo>}
		\frac{
			\stab^{\mathrm{num}}(\triangcat)
		}{
			\gltworplustilde
		}.
	\]
	\noindent
	In the case of a principally polarised abelian surface, descends to:
	\[
		\PSL(2, \ZZ)
		\circlearrowright
		% \HCode{<mo class="MathClass-rel" stretchy="false">⟳</mo>}
		\left\{
		z \in \CC
		\colon
		\mathfrak{Im}(z) > 0
		\right\},
	\]
	coinciding with the standard action of
	the ``modular group'' $\PSL(2, \ZZ)$
	on the upper half-plane via Möbius transformations.
\end{theorem}

\begin{proof}
First, note that the two natural group actions
(from $\aut\left(\triangcat\right)^{\mathrm{num}}$
and $\gltworplustilde$)
on  $\stab^{\mathrm{num}}(\triangcat)$ commute with
each other so that there is an induced action
\[
	\aut\left(\triangcat\right)^{\mathrm{num}}
	\circlearrowright
	% \HCode{<mo class="MathClass-rel" stretchy="false">⟳</mo>}
	\frac{
		\stab^{\mathrm{num}}(\triangcat)
	}{
		\gltworplustilde
	}.
\]

\end{proof}
