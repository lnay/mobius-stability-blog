\section{Main statements}
\begin{theorem}
	There is a group action:
	\[
		\aut\left(\triangcat\right)^{\mathrm{num}}
		\circlearrowright
		%\HCode{<mo class="MathClass-rel" stretchy="false">⟳</mo>}
		\frac{
			\stab^{\mathrm{num}}(\triangcat)
		}{
			\gltworplustilde
		}\cdot
	\]
	Furthermore, it still makes sense to talk about walls on the right-hand side,
	and this group action preserves the set of points on walls, possibly permuting the walls for a given group element.
\end{theorem}
\begin{proof}
	Note that the two natural group actions
	(from $\aut\left(\triangcat\right)^{\mathrm{num}}$
	and $\gltworplustilde$)
	on  $\stab^{\mathrm{num}}(\triangcat)$ commute with
	each other so that there is an induced action on the
	quotient as described in the statement of the theorem.

	Since  $\gltworplustilde$ preserves semi-stable objects, and only modifies the central charge, if a stability condition point is on a wall, it is still on a wall after any action by some $g\in\gltworplustilde$.
\end{proof}

\begin{theorem}
	\label{thm:modular_group_action}
	The group action:
	\[
		\aut\left(\triangcat\right)^{\mathrm{num}}
		\circlearrowright
		%\HCode{<mo class="MathClass-rel" stretchy="false">⟳</mo>}
		\frac{
			\stab^{\mathrm{num}}(\triangcat)
		}{
			\gltworplustilde
		},
	\]
	\noindent
	in the case of a principally polarised abelian surface, descends to:
	\[
		\PSL(2, \ZZ)
		\circlearrowright
		% \HCode{<mo class="MathClass-rel" stretchy="false">⟳</mo>}
		\left\{
		z \in \CC
		\colon
		\mathfrak{Im}(z) > 0
		\right\},
	\]
	coinciding with the standard action of
	the ``modular group'' $\PSL(2, \ZZ)$
	on the upper half-plane via Möbius transformations.
\end{theorem}

\begin{proof}
	For now, refer to Section 2.4 from my thesis: \url{https://era.ed.ac.uk/handle/1842/43553}.
\end{proof}

\begin{example}[Standard Fourier-Mukai Transform]
	Consider the standard Fourier-Mukai transform on a general principally polarised abelian surface, using the Picard bundle\footnote{This is the transform which takes flat line bundles to skyscraper sheaves shifted twice.}.
	The corresponding element of $\PSL(2, \ZZ)$ is
		{\footnotesize
			\(
			\left[
				\begin{pmatrix}
					0  & 1 \\
					-1 & 0
				\end{pmatrix}
				\right]
			\)},
	which corresponds to the Möbius transformation
	\(
	z \mapsto -\frac{1}{z}
	\).

\end{example}
